local LucideManager = {Module = nil, Loaded = false}
local function LoadLucide()
    if LucideManager.Loaded then return LucideManager.Module end
    local content = game:HttpGet("https://raw.githubusercontent.com/deividcomsono/lucide-roblox-direct/refs/heads/main/source.lua")
    if not content or content == "" then return nil end
    local func, err = loadstring(content)
    if not func then return nil end
    local module = func()
    if module then
        LucideManager.Module = module
        LucideManager.Loaded = true
        return module
    end
    return nil
end
local function GetIcon(iconName, size, color)
    local lucide = LoadLucide()
    if not lucide then return nil end
    local icon = lucide.GetAsset(iconName)
    if not icon then return nil end
    local img = Instance.new("ImageLabel")
    img.Image = icon.Url
    img.Size = size or UDim2.new(0, 20, 0, 20)
    img.BackgroundTransparency = 1
    img.ImageColor3 = color or Color3.fromRGB(230, 232, 240)
    img.ScaleType = Enum.ScaleType.Fit
    if icon.ImageRectOffset then
        img.ImageRectOffset = icon.ImageRectOffset
        img.ImageRectSize = icon.ImageRectSize
    end
    return img
end
local function create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do inst[k] = v end
    return inst
end
local function corner(radius, parent)
    local c = create("UICorner", {CornerRadius = UDim.new(0, radius or 8)})

    local s = create("UIStroke", {Color = color or Color3.fromRGB(60, 65, 80), Thickness = thickness or 1, Transparency = 0.4})

end

local isOrbDragging = false
local orbDragStart = nil
local orbStartPos = nil
local orbDragInput = nil

orbBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isOrbDragging = false
        orbDragStart = Vector2.new(input.Position.X, input.Position.Y)
        orbDragOffset = (orbFrame.AbsolutePosition + orbFrame.AbsoluteSize / 2) - orbDragStart
        orbDragInput = input
    end
end)

v2.InputChanged:Connect(function(input)
    if input == orbDragInput and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local pos2 = Vector2.new(input.Position.X, input.Position.Y)
        local delta = pos2 - orbDragStart
        if math.abs(delta.X) > 5 or math.abs(delta.Y) > 5 then
            isOrbDragging = true
        end
        if isOrbDragging then
            local newPos = pos2 + orbDragOffset
            orbFrame.Position = UDim2.new(0, newPos.X, 0, newPos.Y)
        end
    end
end)

v2.InputEnded:Connect(function(input)
    if input == orbDragInput then
        orbDragInput = nil
        if not isOrbDragging then
            main.Visible = true
            orbFrame.Visible = false
        end
        isOrbDragging = false
    end
end)

local function makeToggle(parent, initialState, callback)
    local toggleBg = create("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.new(0, 48, 0, 26), BackgroundColor3 = initialState and theme.accent or theme.surfaceLight, BackgroundTransparency = 0.2, BorderSizePixel = 0, Text = "", ZIndex = 5})
    corner(13, toggleBg)
    local knob = create("Frame", {AnchorPoint = Vector2.new(0, 0.5), Position = initialState and UDim2.new(1, -22, 0.5, 0) or UDim2.new(0, 4, 0.5, 0), Size = UDim2.new(0, 18, 0, 18), BackgroundColor3 = Color3.fromRGB(255,255,255), BorderSizePixel = 0, ZIndex = 6})
    corner(9, knob)
    knob.Parent = toggleBg
    toggleBg.Parent = parent
    local state = initialState
    toggleBg.MouseButton1Click:Connect(function()
        state = not state
        v4:Create(toggleBg, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = state and theme.accent or theme.surfaceLight}):Play()
        v4:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = state and UDim2.new(1, -22, 0.5, 0) or UDim2.new(0, 4, 0.5, 0)}):Play()
        local ok, err = pcall(function() callback(state) end)
        if not ok then
            warn("[DeltaUI] " .. tostring(err))
        end
    end)
    return toggleBg, function() return state end
end

local function makeDropdown(parent, options, defaultIndex, callback)
    local current = options[defaultIndex] or options[1]
    local btn = create("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.new(0, 120, 0, 32), BackgroundColor3 = theme.surfaceLight, BackgroundTransparency = 0.2, Text = "", BorderSizePixel = 0, ZIndex = 5})
    corner(8, btn)
    local txt = create("TextLabel", {Position = UDim2.new(0, 10, 0, 0), Size = UDim2.new(1, -30, 1, 0), BackgroundTransparency = 1, Text = current, TextColor3 = theme.text, TextSize = 12, Font = Enum.Font.GothamBold, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 6})
    txt.Parent = btn
    local arrow = GetIcon("chevron-down", UDim2.new(0, 14, 0, 14), theme.textDim)
    if arrow then
        arrow.Position = UDim2.new(1, -20, 0.5, -7)
        arrow.Parent = btn
    end
    btn.Parent = parent
    local dropdownOpen = false
    local listFrame = nil
    btn.MouseButton1Click:Connect(function()
        if dropdownOpen and listFrame then
            listFrame:Destroy()
            listFrame = nil
            dropdownOpen = false
            return
        end
        dropdownOpen = true
        listFrame = create("Frame", {Position = UDim2.new(0, 0, 0, 36), Size = UDim2.new(1, 0, 0, #options * 30), BackgroundColor3 = theme.surfaceLight, BackgroundTransparency = 0.05, BorderSizePixel = 0, ZIndex = 250})
        corner(8, listFrame)
        stroke(theme.border, 1, listFrame)
        listFrame.Parent = btn
        for i, opt in ipairs(options) do
            local optBtn = create("TextButton", {Position = UDim2.new(0, 0, 0, (i-1)*30), Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1, Text = "", ZIndex = 251})
            local optTxt = create("TextLabel", {Position = UDim2.new(0, 10, 0, 0), Size = UDim2.new(1, -20, 1, 0), BackgroundTransparency = 1, Text = opt, TextColor3 = theme.text, TextSize = 12, Font = Enum.Font.Gotham, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 252})
            optTxt.Parent = optBtn
            optBtn.Parent = listFrame
            optBtn.MouseButton1Click:Connect(function()
                current = opt
                txt.Text = current
                if listFrame then
                    listFrame:Destroy()
                    listFrame = nil
                end
                dropdownOpen = false
                local ok, err = pcall(function() callback(current) end)
                if not ok then
                    warn("[DeltaUI] " .. tostring(err))
                end
            end)
        end
    end)
    return btn, function() return current end
end

        local pos = input.Position
        local cardPos = modalCard.AbsolutePosition
        local cardSize = modalCard.AbsoluteSize
        if pos.X < cardPos.X or pos.X > cardPos.X + cardSize.X or pos.Y < cardPos.Y or pos.Y > cardPos.Y + cardSize.Y then
            modalOverlay.Visible = false
        end
    end
end)
addScriptBtn.MouseButton1Click:Connect(function()
    local title = titleInput.Text
    local scriptContent = scriptInput.Text
    if title == "" or title == "Enter Your Title..." then
        return
    end
    if scriptContent == "" or scriptContent == "Enter Your Script..." then
        return
    end
    ensureFolder()
    local fileName = title:gsub("[^%w%s_-]", ""):gsub("%s+", "_")
    if fileName == "" then fileName = "untitled" end
    writefile(saveFolder .. "/" .. fileName .. ".lua", scriptContent)
    modalOverlay.Visible = false
    refreshScriptList(searchInput.Text)
end)
searchInput:GetPropertyChangedSignal("Text"):Connect(function()
    refreshScriptList(searchInput.Text)
end)
task.spawn(function()
    task.wait(0.5)
    refreshScriptList("")
end)
local function switchPage(pageName)
    if pageName == currentPage then return end
    currentPage = pageName
    for _, page in pairs(pages) do
        page.Visible = false
    end
    if pages[pageName] then
        pages[pageName].Visible = true
    end
    bottomBar.Visible = (pageName == "house")
    animateIndicator(navButtons[pageName])
end
for _, name in ipairs(navNames) do
    navButtons[name].MouseButton1Click:Connect(function()
        switchPage(name)
    end)
end
task.spawn(function()
    while true do
        local t = os.date("*t")
        local suffix = t.hour >= 12 and "PM" or "AM"
        local hour = t.hour % 12
        if hour == 0 then hour = 12 end
        local min = string.format("%02d", t.min)
        local timeStr = hour .. ":" .. min .. " " .. suffix
        for _, label in ipairs(allTimeLabels) do
            label.Text = timeStr
        end
        task.wait(60)
    end
end)
task.spawn(function()
    local lastUpdate = 0
    local frameCount = 0
    v5.RenderStepped:Connect(function()
        frameCount = frameCount + 1
        local now = tick()
        if now - lastUpdate >= 1 then
            local fpsText = frameCount .. " FPS"
            local pingText = "0 ms"
            local ping = v6.Network.ServerStatsItem["Data Ping"]
            if ping then
                pingText = math.floor(ping:GetValue()) .. " ms"
            end
            for _, label in ipairs(allPingLabels) do
                label.Text = pingText
            end
            for _, label in ipairs(allFpsLabels) do
                label.Text = fpsText
            end
            frameCount = 0
            lastUpdate = now
        end
    end)
end)
execBtn.MouseButton1Click:Connect(function()
    cleanupOldUI()
    local code = codeBox.Text
    if code and code ~= "" then
        AddLog("> Executing script...", "info")
        local fn, compileErr = loadstring(code)
        if not fn then
            AddLog("[Error] " .. tostring(compileErr), "error")
            return
        end
        fn()
        AddLog("> Execution finished", "info")
    end
end)

local recentErrors = {}
game:GetService("ScriptContext").Error:Connect(function(message, stackTrace, script)
    local msg = tostring(message)
    local now = tick()
    if recentErrors[msg] and (now - recentErrors[msg].time) < 3 then
        recentErrors[msg].count = recentErrors[msg].count + 1
        return
    end
    recentErrors[msg] = {time = now, count = 1}
    if not msg or msg == "" then return end
    if msg:find("is not a valid member of") then return end
    AddLog("[Server] " .. msg, "error")
end)
