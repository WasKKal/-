local LucideManager = {Module = nil, Loaded = false}
local function LoadLucide()
    if LucideManager.Loaded then return LucideManager.Module end
    local content = game:HttpGet("https://raw.githubusercontent.com/deividcomsono/lucide-roblox-direct/refs/heads/main/source.lua")
    if not content or content == "" then return nil end
    local func, err = loadstring(content)
    if not func then return nil end
    local module = func()
    if module then
        LucideManager.Module = module
        LucideManager.Loaded = true
        return module
    end
    return nil
end
local function GetIcon(iconName, size, color)
    local lucide = LoadLucide()
    if not lucide then return nil end
    local icon = lucide.GetAsset(iconName)
    if not icon then return nil end
    local img = Instance.new("ImageLabel")
    img.Image = icon.Url
    img.Size = size or UDim2.new(0, 20, 0, 20)
    img.BackgroundTransparency = 1
    img.ImageColor3 = color or Color3.fromRGB(230, 232, 240)
    img.ScaleType = Enum.ScaleType.Fit
    if icon.ImageRectOffset then
        img.ImageRectOffset = icon.ImageRectOffset
        img.ImageRectSize = icon.ImageRectSize
    end
    return img
end
local function create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do inst[k] = v end
    return inst
end
local function corner(radius, parent)
    local c = create("UICorner", {CornerRadius = UDim.new(0, radius or 8)})
    c.Parent = parent
    return c
end
local function stroke(color, thickness, parent)
    local s = create("UIStroke", {Color = color or Color3.fromRGB(60, 65, 80), Thickness = thickness or 1, Transparency = 0.4})
    s.Parent = parent
    return s
end

local settingsData = {language = "en", uiRefs = {}}
local configFile = "DeltaPlus/config.json"

local function loadConfig()
    if not isfile(configFile) then return {} end
    local ok, content = pcall(readfile, configFile)
    if not ok or not content then return {} end
    local ok2, data = pcall(function() return game:GetService("HttpService"):JSONDecode(content) end)
    if ok2 and type(data) == "table" then return data end
    return {}
end

local function saveConfig(data)
    if not isfolder("DeltaPlus") then makefolder("DeltaPlus") end
    local ok, json = pcall(function() return game:GetService("HttpService"):JSONEncode(data) end)
    if ok and json then
        pcall(function() writefile(configFile, json) end)
    end
end

local function t(key)
    local lang = settingsData.language or "en"
    local entry = translations[key]
    if entry then
        return entry[lang] or entry.en or key
    end
    return key
end

local function makeActionButton(text, parent, callback)
    local btn = create("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.new(0, 110, 0, 32), BackgroundColor3 = theme.accent, BackgroundTransparency = 0.15, Text = "", BorderSizePixel = 0, ZIndex = 5})
    corner(8, btn)
    local txt = create("TextLabel", {Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Text = text, TextColor3 = Color3.fromRGB(255,255,255), TextSize = 12, Font = Enum.Font.SourceSansBold, ZIndex = 6})
    txt.Parent = btn
    btn.Parent = parent
    btn.MouseButton1Click:Connect(function()
        local ok, err = pcall(callback)
        if not ok then
            warn("[DeltaUI] " .. tostring(err))
        end
    end)
    return btn
end

local function makeToggle(parent, initialState, callback)
    local toggleBg = create("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.new(0, 48, 0, 26), BackgroundColor3 = initialState and theme.accent or theme.surfaceLight, BackgroundTransparency = 0.2, BorderSizePixel = 0, Text = "", ZIndex = 5})
    corner(13, toggleBg)
    local knob = create("Frame", {AnchorPoint = Vector2.new(0, 0.5), Position = initialState and UDim2.new(1, -22, 0.5, 0) or UDim2.new(0, 4, 0.5, 0), Size = UDim2.new(0, 18, 0, 18), BackgroundColor3 = Color3.fromRGB(255,255,255), BorderSizePixel = 0, ZIndex = 6})
    corner(9, knob)
    knob.Parent = toggleBg
    toggleBg.Parent = parent
    local state = initialState
    toggleBg.MouseButton1Click:Connect(function()
        state = not state
        v4:Create(toggleBg, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = state and theme.accent or theme.surfaceLight}):Play()
        v4:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = state and UDim2.new(1, -22, 0.5, 0) or UDim2.new(0, 4, 0.5, 0)}):Play()
        local ok, err = pcall(function() callback(state) end)
        if not ok then
            warn("[DeltaUI] " .. tostring(err))
        end
    end)
    return toggleBg, function() return state end
end

local function makeDropdown(parent, options, defaultIndex, callback)
    local current = options[defaultIndex] or options[1]
    local btn = create("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.new(0, 120, 0, 32), BackgroundColor3 = theme.surfaceLight, BackgroundTransparency = 0.2, Text = "", BorderSizePixel = 0, ZIndex = 5})
    corner(8, btn)
    local txt = create("TextLabel", {Position = UDim2.new(0, 10, 0, 0), Size = UDim2.new(1, -30, 1, 0), BackgroundTransparency = 1, Text = current, TextColor3 = theme.text, TextSize = 12, Font = Enum.Font.SourceSansBold, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 6})
    txt.Parent = btn
    local arrow = GetIcon("chevron-down", UDim2.new(0, 14, 0, 14), theme.textDim)
    if arrow then
        arrow.Position = UDim2.new(1, -20, 0.5, -7)
        arrow.Parent = btn
    end
    btn.Parent = parent
    local dropdownOpen = false
    local listFrame = nil
    btn.MouseButton1Click:Connect(function()
        if dropdownOpen and listFrame then
            listFrame:Destroy()
            listFrame = nil
            dropdownOpen = false
            return
        end
        dropdownOpen = true
        listFrame = create("Frame", {
            Position = UDim2.new(0, btn.AbsolutePosition.X, 0, btn.AbsolutePosition.Y + 36),
            Size = UDim2.new(0, btn.AbsoluteSize.X, 0, #options * 30),
            BackgroundColor3 = theme.surfaceLight,
            BackgroundTransparency = 0.05,
            BorderSizePixel = 0,
            ZIndex = 999,
            ClipsDescendants = false
        })
        corner(8, listFrame)
        stroke(theme.border, 1, listFrame)
        listFrame.Parent = screenGui
        for i, opt in ipairs(options) do
            local optBtn = create("TextButton", {Position = UDim2.new(0, 0, 0, (i-1)*30), Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1, Text = "", ZIndex = 1000})
            local optTxt = create("TextLabel", {Position = UDim2.new(0, 10, 0, 0), Size = UDim2.new(1, -20, 1, 0), BackgroundTransparency = 1, Text = opt, TextColor3 = theme.text, TextSize = 12, Font = Enum.Font.SourceSans, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 1001})
            optTxt.Parent = optBtn
            optBtn.Parent = listFrame
            optBtn.MouseButton1Click:Connect(function()
                current = opt
                txt.Text = current
                if listFrame then
                    listFrame:Destroy()
                    listFrame = nil
                end
                dropdownOpen = false
                local ok, err = pcall(function() callback(current) end)
                if not ok then
                    warn("[DeltaUI] " .. tostring(err))
                end
            end)
        end
    end)
    return btn, function() return current end
end

local consoleEnabled = true
local maxLogEntries = 500
local logEntryCount = 0
local pendingLogs = {}
local logProcessing = false

local function AddLog(message, level)
    if not consoleEnabled then
        return
    end
    table.insert(pendingLogs, {msg = tostring(message), lvl = level})
    if logProcessing then return end
    logProcessing = true
    task.defer(function()
        while #pendingLogs > 0 do
            local batch = {}
            for i = 1, math.min(10, #pendingLogs) do
                batch[i] = table.remove(pendingLogs, 1)
            end
            for _, log in ipairs(batch) do
                local color = theme.text
                if log.lvl == "error" then color = theme.red
                elseif log.lvl == "warn" then color = theme.warn
                elseif log.lvl == "info" then color = theme.textDim end
                local entry = create("TextLabel", {
                    Size = UDim2.new(1, 0, 0, 18),
                    BackgroundTransparency = 1,
                    Text = log.msg,
                    TextColor3 = color,
                    TextSize = 11,
                    Font = Enum.Font.Code,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    TextYAlignment = Enum.TextYAlignment.Center,
                    ZIndex = 4
                })
                entry.Parent = consoleScroll
                logEntryCount = logEntryCount + 1
                if logEntryCount > maxLogEntries then
                    local first = consoleScroll:FindFirstChildWhichIsA("TextLabel")
                    if first then
                        first:Destroy()
                        logEntryCount = logEntryCount - 1
                    end
                end
            end
            consoleScroll.CanvasPosition = Vector2.new(0, consoleScroll.AbsoluteCanvasSize.Y)
            if #pendingLogs > 0 then
                task.wait(0.05)
            end
        end
        logProcessing = false
    end)
end

local function getClipboardContent()
    local methods = {
        function() return syn_clipboard_get and syn_clipboard_get() end,
        function() return getclipboard and getclipboard() end,
        function() return KRNL_CLIPBOARD and KRNL_CLIPBOARD.get and KRNL_CLIPBOARD.get() end,
        function() return Clipboard and Clipboard.get and Clipboard.get() end,
    }
    for _, method in ipairs(methods) do
        local ok, result = pcall(method)
        if ok and result and result ~= "" then
            return result
        end
    end
    return nil
end

local function refreshScriptList(filter)
    for _, child in pairs(scriptListScroll:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end

    local ok, files = pcall(listfiles, saveFolder)
    if not ok or not files then return end
    for _, filePath in ipairs(files) do
        local name = filePath:match("([^/\\]+)%.lua$") or filePath:match("([^/\\]+)$")
        if not name then
        elseif filter and filter ~= "" and not name:lower():find(filter:lower()) then
        else
            local item = create("Frame", {
                Size = UDim2.new(1, 0, 0, 44),
                BackgroundColor3 = theme.surface,
                BackgroundTransparency = 0.15,
                BorderSizePixel = 0,

                Position = UDim2.new(0, 14, 0, 0),
                Size = UDim2.new(0.5, 0, 1, 0),
                BackgroundTransparency = 1,
                Text = name,
                TextColor3 = theme.text,
                TextSize = 13,
                Font = Enum.Font.SourceSansBold,
                TextXAlignment = Enum.TextXAlignment.Left,
                TextYAlignment = Enum.TextYAlignment.Center,
                ZIndex = 5
            })
            itemTitle.Parent = item
            local delBtn = create("TextButton", {
                AnchorPoint = Vector2.new(1, 0.5),
                Position = UDim2.new(1, -96, 0.5, 0),
                Size = UDim2.new(0, 80, 0, 28),
                BackgroundColor3 = theme.surfaceLight,
                BackgroundTransparency = 0.2,
                Text = "",

            })

            local delText = create("TextLabel", {Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Text = t("DELETE"), TextColor3 = theme.text, TextSize = 11, Font = Enum.Font.SourceSansBold, ZIndex = 6})

                AnchorPoint = Vector2.new(1, 0.5),
                Position = UDim2.new(1, -8, 0.5, 0),
                Size = UDim2.new(0, 80, 0, 28),

local function switchPage(pageName)
    if pageName == currentPage then return end
    currentPage = pageName
    for _, page in pairs(pages) do
        page.Visible = false
    end
    if pages[pageName] then
        pages[pageName].Visible = true
    end
    bottomBar.Visible = (pageName == "house")
    animateIndicator(navButtons[pageName])
end

task.spawn(function()
    while true do
        local ok, timeTable = pcall(function() return os.date("*t") end)
        if ok and timeTable then
            local suffix = timeTable.hour >= 12 and "PM" or "AM"
            local hour = timeTable.hour % 12
            if hour == 0 then hour = 12 end
            local min = string.format("%02d", timeTable.min)
            local timeStr = hour .. ":" .. min .. " " .. suffix
            for _, label in ipairs(allTimeLabels) do
                if label and label.Parent then
                    label.Text = timeStr
                end
            end
        end
        task.wait(60)
    end
end)

local lastUpdate = 0
local frameCount = 0
v5.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    local now = tick()
    if now - lastUpdate >= 1 then
        local fpsText = frameCount .. " FPS"
        local pingText = "0 ms"
        local ok, ping = pcall(function()
            return v6.Network.ServerStatsItem["Data Ping"]
        end)
        if ok and ping then
            local ok2, val = pcall(function() return ping:GetValue() end)
            if ok2 and val then
                pingText = math.floor(val) .. " ms"
            end
        end
        for _, label in ipairs(allPingLabels) do
            if label and label.Parent then
                label.Text = pingText
            end
        end
        for _, label in ipairs(allFpsLabels) do
            if label and label.Parent then
                label.Text = fpsText
            end
        end
        frameCount = 0
        lastUpdate = now
    end
end)

local function applyConfig()
    local cfg = loadConfig()
    if cfg.language then
        settingsData.language = cfg.language
    end
    if cfg.fpsCap and setfpscap then
        local cap = cfg.fpsCap == "30 FPS" and 30 or cfg.fpsCap == "60 FPS" and 60 or cfg.fpsCap == "120 FPS" and 120 or cfg.fpsCap == "240 FPS" and 240 or cfg.fpsCap == "360 FPS" and 360 or 0
        setfpscap(cap)
    end
    if cfg.iconSize then
        local sz = cfg.iconSize == "Small" and 25 or cfg.iconSize == "Medium" and 35 or 45
        orbFrame.Size = UDim2.new(0, sz, 0, sz)
    end
    if cfg.iconShape then
        for _, c in pairs(orbFrame:GetChildren()) do
            if c:IsA("UICorner") then
                c:Destroy()
            end
        end
        if cfg.iconShape == "Circle" then
            local sz = orbFrame.Size.X.Offset
            corner(sz/2, orbFrame)
        end
    end
    if cfg.disableRobux then
        getRobuxState()
    end
    if cfg.verifyTeleports then
        getVerifyState()
    end
    if cfg.antiAfk then
        getAntiAfkState()
    end
    if cfg.console == false then
        getConsoleState()
    end
    if cfg.autoExec then
        getAutoExecState()
    end
end
applyConfig()

local logService = game:GetService("LogService")
local lastLogTime = 0
local logDedup = {}
logService.MessageOut:Connect(function(msg, msgtype)
    local now = tick()
    local key = tostring(msg)
    if logDedup[key] and (now - logDedup[key]) < 2 then
        return
    end
    logDedup[key] = now
    local level = "info"
    if msgtype == Enum.MessageType.MessageWarning then level = "warn"
    elseif msgtype == Enum.MessageType.MessageError then level = "error" end
    AddLog(msg, level)
end)
